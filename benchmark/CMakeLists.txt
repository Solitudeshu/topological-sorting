cmake_minimum_required(VERSION 3.16)
project(topological_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(topo_bench src/runner.cpp)
target_include_directories(topo_bench PRIVATE include)
add_executable(topo_tests tests/test_topo.cpp)
target_include_directories(topo_tests PRIVATE include)
if(MSVC)
  target_compile_options(topo_bench PRIVATE /O2 /W4)
  target_compile_options(topo_tests PRIVATE /O2 /W4)
else()
  target_compile_options(topo_bench PRIVATE -O3 -Wall -Wextra -Wpedantic)
  target_compile_options(topo_tests PRIVATE -O2 -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME topo_correctness COMMAND topo_tests)
